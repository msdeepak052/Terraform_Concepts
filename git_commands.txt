--------------------------------------------------------------------
Pushing to just one remote (rarely needed, dual-push covers normal use)
--------------------------------------------------------------------
git push gitlab terraform-associate-hashicorp-004     # push to GitLab only
git -c remote.origin.pushurl=https://github.com/msdeepak052/Terraform_Concepts.git push origin terraform-associate-hashicorp-004   # push to GitHub only, bypassing the dual pushurl for this one command

--------------------------------------------------------------------
Checking state
--------------------------------------------------------------------
git remote -v                                  # list configured remotes
git branch -a                                  # list local + remote-tracking branches
git branch --show-current                      # current branch name
git log --oneline -10                          # recent commit history
git log --oneline --all --graph -15            # full graph across all branches
git ls-remote --heads <repo-url>                # inspect a remote's branches without cloning

--------------------------------------------------------------------
One-time setup steps already performed (for reference / re-doing elsewhere)
--------------------------------------------------------------------
git init
git remote add origin https://github.com/msdeepak052/Terraform_Concepts.git
git remote add gitlab https://gitlab.com/terraform-hashicorp-associate/terraform-hashicorp-004-complete.git
git fetch origin

git checkout --orphan terraform-associate-hashicorp-004   # branch with no shared history
git add .gitignore notes/
git commit -m "Add Terraform Associate (003/004) exam-prep notes"

git push -u origin terraform-associate-hashicorp-004      # first push + set upstream
git push -u gitlab terraform-associate-hashicorp-004      # push to GitLab too

# after pushing to gitlab with -u, upstream tracking pointed at gitlab;
# reset it back to origin so plain push/pull default to GitHub:
git branch --set-upstream-to=origin/terraform-associate-hashicorp-004 terraform-associate-hashicorp-004

# make a single `git push` reach both GitHub and GitLab: add a second
# push URL onto origin (fetch URL stays GitHub-only, untouched)
git remote set-url --add --push origin https://github.com/msdeepak052/Terraform_Concepts.git
git remote set-url --add --push origin https://gitlab.com/terraform-hashicorp-associate/terraform-hashicorp-004-complete.git
git remote -v   # should now show origin with TWO push lines (github + gitlab)

--------------------------------------------------------------------
Amending the last commit (only if it hasn't been relied on elsewhere)
--------------------------------------------------------------------
git commit --amend -m "new message"
git push --force-with-lease origin terraform-associate-hashicorp-004   # required after amend, since history rewrote

--------------------------------------------------------------------
Notes
--------------------------------------------------------------------
- Never use `git push --force` (no lease) or `git reset --hard` without
  checking `git status` first -- both can silently discard work.
- `--force-with-lease` refuses to push if someone else updated the
  remote branch since your last fetch, which plain `--force` would not catch.
